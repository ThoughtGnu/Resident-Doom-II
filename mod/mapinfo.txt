	StatusBarClass = "REStatusBar"
	AddEventHandlers = "REContinueHandler", "RECameraHandler", "REGameHandler", "RELightingHandler", "REInputBridge", "REViewHandler",

	statscreen_single = "REStatusScreen"
	// Title screen (mod/hires/TITLEPIC.png): stay on it instead of moving on
	// to Doom II's demos, which don't play back with tank controls.
	TitleTime = 100000
	// Quitting: one plain question (language.txt REQUITMSG) instead of Doom's
	// joke lines, and no red ENDOOM text screen afterwards (no such lump).
	QuitMessages = "$REQUITMSG"
	Endoom = "RENOENDO"
	// Every yes/no question box gets a dark layer behind it (RE_Title.zs).
	MessageBoxClass = "REMessageBox"
}
